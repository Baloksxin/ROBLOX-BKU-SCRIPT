local function setCharacterAnimationFrozen(char, frozen)
	if not char then return end
	local humanoid = char:FindFirstChildOfClass("Humanoid")
	if humanoid then
		local animator = humanoid:FindFirstChildOfClass("Animator")
		if animator then
			if frozen then
				for _, track in ipairs(animator:GetPlayingAnimationTracks()) do
					cachedAnimationSpeeds[track] = track.Speed
					track:AdjustSpeed(0)
				end
			else
				for _, track in ipairs(animator:GetPlayingAnimationTracks()) do
					if cachedAnimationSpeeds[track] then
						track:AdjustSpeed(cachedAnimationSpeeds[track])
					else
						track:AdjustSpeed(1)
					end
				end
				table.clear(cachedAnimationSpeeds)
				humanoid:ChangeState(Enum.HumanoidStateType.GettingUp)
			end
		end
	end
end

local function removeCreatorESP(player)

local function startRotation()
	if rotationConnection then
		rotationConnection:Disconnect()
		rotationConnection = nil
	end

	if not RotationConfig.Enabled then
		if LPlr.Character then
			local humanoid = LPlr.Character:FindFirstChildOfClass("Humanoid")
			if humanoid then
				humanoid.AutoRotate = true
			end
		end
		return
	end

	if LPlr.Character then
		local humanoid = LPlr.Character:FindFirstChildOfClass("Humanoid")
		if humanoid then
			humanoid.AutoRotate = false
		end
	end

	rotationConnection = RS.RenderStepped:Connect(function()
		if not RotationConfig.Enabled then
			return
		end
		local char = LPlr.Character
		if not char then
			return
		end
		local hrp = char:FindFirstChild("HumanoidRootPart")
		if not hrp then
			return
		end
		local currentCFrame = hrp.CFrame
		local rotationAngle = math.rad(RotationConfig.Speed * RotationConfig.Direction)
		local newCFrame = currentCFrame * CFrame.Angles(0, rotationAngle, 0)
		hrp.CFrame = newCFrame
	end)
end

local antiSlipperyEnabled = false
local antiSlipperyConnection = nil

local function antiSlippery()
	local char = LPlr.Character
	if not char then
		return
	end
	local humanoid = char:FindFirstChildOfClass("Humanoid")
	local rootPart = char:FindFirstChild("HumanoidRootPart")
	if not humanoid or not rootPart then
		return
	end
	if humanoid.Health <= 0 then
		return
	end
	local currentVel = rootPart.AssemblyLinearVelocity
	local moveDir = humanoid.MoveDirection
	local walkSpeed = humanoid.WalkSpeed
	if walkSpeed <= 0 or walkSpeed > 200 then
		return
	end
	if moveDir.Magnitude < 0.05 then
		if math.abs(currentVel.X) > 0.01 or math.abs(currentVel.Z) > 0.01 then
			rootPart.AssemblyLinearVelocity = Vector3.new(0, currentVel.Y, 0)
		end
	else
		local targetHorizontal = moveDir.Unit * walkSpeed
		rootPart.AssemblyLinearVelocity = Vector3.new(targetHorizontal.X, currentVel.Y, targetHorizontal.Z)
	end
end

local function enableAntiSlippery()
	if antiSlipperyEnabled then
		return
	end
	if not LPlr.Character or not LPlr.Character:FindFirstChild("HumanoidRootPart") then
		task.wait(0.5)
	end
	antiSlipperyConnection = RS.Heartbeat:Connect(antiSlippery)
	antiSlipperyEnabled = true
end

local function disableAntiSlippery()
	if not antiSlipperyEnabled then
		return
	end
	if antiSlipperyConnection then
		antiSlipperyConnection:Disconnect()
		antiSlipperyConnection = nil
	end
	antiSlipperyEnabled = false
end

local function IsMonster(v)
	if v:IsA("Model") and v:FindFirstChildOfClass("Humanoid") then
		local isPlayer = false
		for _, p in pairs(Players:GetPlayers()) do
			if p.Character == v then
				isPlayer = true
				break
			end
		end
		if not isPlayer then
			local nameLower = v.Name
			if nameLower == "slender" or nameLower == "Claw" or nameLower == "Crawly" or nameLower == "SawRunner" then
				return true
			end
		end
	end
	return false
end

local function hideBody(model)
	if model:IsA("Model") or model:IsA("BasePart") then
		local nameLower = model.Name:lower()
		local humanoid = model:IsA("Model") and model:FindFirstChildOfClass("Humanoid") or nil
		
		local isDeadBody = nameLower:find("dead") or 
		                   nameLower:find("corpse") or 
		                   nameLower:find("ragdoll") or 
		                   nameLower:find("chimpragdoll") or 
		                   (humanoid and humanoid.Health <= 0)
		
		if isDeadBody then
			task.defer(function()
				if model and model.Parent then
					model.Archivable = true
					model.Parent = nil
				end
			end)
		end
	end
end

local function IsValidGroundItem(obj)
	local allPlayers = Players:GetPlayers()
	for i = 1, #allPlayers do
		local p = allPlayers[i]
		if p.Character and obj:IsDescendantOf(p.Character) then
			return false
		end
		local bp = p:FindFirstChild("Backpack")
		if bp and obj:IsDescendantOf(bp) then
			return false
		end
	end
	if obj:IsA("BasePart") then
		if obj.Transparency > 0.5 then
			return false
		end
		if obj.Position.Y < -50 then
			return false
		end
	else
		local cp = obj:GetPivot().Position
		if cp.Y < -50 then
			return false
		end
	end
	return true
end

local function PlayerHasLootToSell()
	local char = LPlr.Character
	local bp = LPlr:FindFirstChild("Backpack")
	if bp then
		for _, tool in pairs(bp:GetChildren()) do
			if tool:IsA("Tool") and ItemPriority[tool.Name] then
				return true
			end
		end
	end
	if char then
		for _, tool in pairs(char:GetChildren()) do
			if tool:IsA("Tool") and ItemPriority[tool.Name] then
				return true
			end
		end
	end
	return false
end

local function FindBestItem()
	local bestItem = nil
	local bestPriority = 0
	local scrapsFolder = workspace:FindFirstChild("Scraps")
	if not scrapsFolder then
		return nil
	end
	for _, v in pairs(scrapsFolder:GetChildren()) do
		if (v:IsA("BasePart") or v:IsA("Model")) and IsValidGroundItem(v) then
			local priority = ItemPriority[v.Name] or 0
			if priority > bestPriority then
				bestPriority = priority
				bestItem = v
			end
		end
	end
	return bestItem
end

local function ApplyObjectESP(v)
	if not v or not v.Parent then
		return
	end
	if v.Name == "ScrapRecycler" and (v:IsA("BasePart") or v:IsA("Model")) then
		if Config.RecyclerESP then
			CreateRecyclerESP(v)
		end
	elseif v.Name == "Card" and v:IsA("MeshPart") then
		if Config.CardESP and IsValidGroundItem(v) then
			CreateCardESP(v)
			if not table.find(ActiveCards, v) then
				table.insert(ActiveCards, v)
			end
		end
	elseif Config.MonsterESP and IsMonster(v) then
		CreateMonsterESP(v)
	end
end

local function ApplyScrapESPOnly(v)
	if Config.ScrapESP and (v:IsA("BasePart") or v:IsA("Model")) then
		CreateScrapESP(v)
	end
end

RS.RenderStepped:Connect(function()
	if not Config.PlayerESP then
		return
	end
	for _, player in pairs(Players:GetPlayers()) do
		if player == LPlr or not player.Character or not player.Character:FindFirstChild("HumanoidRootPart") then
			if ESPObjects[player] then
				ESPObjects[player].Line.Visible = false
				ESPObjects[player].Text.Visible = false
			end
			continue
		end
		if not ESPObjects[player] then
			ESPObjects[player] = CreateESP(player)
		end
		local obj = ESPObjects[player]
		local hrp = player.Character.HumanoidRootPart
		local screenPos, onScreen = Cam:WorldToViewportPoint(hrp.Position)
		local teamColor = (player.Team and player.TeamColor.Color) or Color3.fromRGB(0, 255, 0)

		if Config.PlayerHighlight then
			if not player.Character:FindFirstChild("PlayerHighlight") then
				local hl = Instance.new("Highlight")
				hl.Name = "PlayerHighlight"
				hl.FillColor = (player.Team and player.TeamColor.Color) or Color3.fromRGB(0, 255, 0)
				hl.OutlineColor = Color3.fromRGB(255, 255, 255)
				hl.Parent = player.Character

				player:GetPropertyChangedSignal("Team"):Connect(function()
					if hl and hl.Parent then
						hl.FillColor = (player.Team and player.TeamColor.Color) or Color3.fromRGB(0, 255, 0)
					end
				end)
			end
		else
			if player.Character:FindFirstChild("PlayerHighlight") then
				player.Character.PlayerHighlight:Destroy()
			end
		end

		if onScreen then
			if Config.ShowTracers then
				obj.Line.From = Vector2.new(Cam.ViewportSize.X / 2, Cam.ViewportSize.Y)
				obj.Line.To = Vector2.new(screenPos.X, screenPos.Y)
				obj.Line.Color = teamColor
				obj.Line.Visible = true
			else
				obj.Line.Visible = false
			end
			if Config.ShowNames then
				local head = player.Character:FindFirstChild("Head")
				local textTarget = head and (head.Position + Vector3.new(0, 0.8, 0)) or (hrp.Position + Vector3.new(0, 3, 0))
				local textPos, textOnScreen = Cam:WorldToViewportPoint(textTarget)
				if textOnScreen then
					obj.Text.Position = Vector2.new(textPos.X, textPos.Y)
				else
					obj.Text.Position = Vector2.new(screenPos.X, screenPos.Y - 20)
				end
				obj.Text.Text = player.DisplayName
				obj.Text.Color = teamColor
				obj.Text.Visible = true
			else
				obj.Text.Visible = false
			end
		else
			obj.Line.Visible = false
			obj.Text.Visible = false
		end
	end
end)

RS.Stepped:Connect(function()
	if Config.Noclip and LPlr.Character then
		for _, v in ipairs(LPlr.Character:GetDescendants()) do
			if v:IsA("BasePart") then
				if originalNoclipCollisions[v] == nil then
					originalNoclipCollisions[v] = v.CanCollide
				end
				v.CanCollide = false
			end
		end
	end
	if Config.AnchorOnly and LPlr.Character then
		local hrp = LPlr.Character:FindFirstChild("HumanoidRootPart")
		if hrp then
			hrp.Anchored = true
		end
	end
	if Config.FreezeSelf and LPlr.Character then
		local hrp = LPlr.Character:FindFirstChild("HumanoidRootPart")
		if hrp then
			hrp.Anchored = true
		end
		local humanoid = LPlr.Character:FindFirstChildOfClass("Humanoid")
		if humanoid then
			local animator = humanoid:FindFirstChildOfClass("Animator")
			if animator then
				for _, track in ipairs(animator:GetPlayingAnimationTracks()) do
					if track.Speed ~= 0 then
						cachedAnimationSpeeds[track] = track.Speed
						track:AdjustSpeed(0)
					end
				end
			end
		end
	end
end)
