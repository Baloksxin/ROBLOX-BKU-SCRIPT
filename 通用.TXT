local fovConnection = nil
local function LockFOV()
    if fovConnection then
        fovConnection:Disconnect()
    end
    if Cam then
        fovConnection = Cam:GetPropertyChangedSignal("FieldOfView"):Connect(function()
            if Config.FOVEn and Cam.FieldOfView ~= Config.FOV then
                Cam.FieldOfView = Config.FOV
            end
        end)
    end
end

local function setCharacterAnimationFrozen(char, frozen)
    if not char then return end
    local humanoid = char:FindFirstChildOfClass("Humanoid")
    if humanoid then
        local animator = humanoid:FindFirstChildOfClass("Animator")
        if animator then
            if frozen then
                for _, track in ipairs(animator:GetPlayingAnimationTracks()) do
                    cachedAnimationSpeeds[track] = track.Speed
                    track:AdjustSpeed(0)
                end
            else
                for _, track in ipairs(animator:GetPlayingAnimationTracks()) do
                    if cachedAnimationSpeeds[track] then
                        track:AdjustSpeed(cachedAnimationSpeeds[track])
                    else
                        track:AdjustSpeed(1)
                    end
                end
                table.clear(cachedAnimationSpeeds)
                humanoid:ChangeState(Enum.HumanoidStateType.GettingUp)
            end
        end
    end
end

LPlr.CharacterAdded:Connect(function()
    table.clear(originalNoclipCollisions)
    table.clear(cachedAnimationSpeeds)
end)

local tRotation = Window:MakeTab({ Name = "旋转", Icon = "rbxassetid://77348087165494" })

local RotationConfig = {
    Enabled = false,
    Speed = 1,
    Direction = 1,
}

local rotationConnection = nil

local function startRotation()
    if rotationConnection then
        rotationConnection:Disconnect()
        rotationConnection = nil
    end

    if not RotationConfig.Enabled then
        if LPlr.Character then
            local humanoid = LPlr.Character:FindFirstChildOfClass("Humanoid")
            if humanoid then
                humanoid.AutoRotate = true
            end
        end
        return
    end

    if LPlr.Character then
        local humanoid = LPlr.Character:FindFirstChildOfClass("Humanoid")
        if humanoid then
            humanoid.AutoRotate = false
        end
    end

    rotationConnection = RS.RenderStepped:Connect(function()
        if not RotationConfig.Enabled then
            return
        end

        local char = LPlr.Character
        if not char then
            return
        end

        local hrp = char:FindFirstChild("HumanoidRootPart")
        if not hrp then
            return
        end

        local currentCFrame = hrp.CFrame
        local rotationAngle = math.rad(RotationConfig.Speed * RotationConfig.Direction)
        local newCFrame = currentCFrame * CFrame.Angles(0, rotationAngle, 0)
        hrp.CFrame = newCFrame
    end)
end

LPlr.CharacterAdded:Connect(function(char)
    task.wait(0.5)
    if RotationConfig.Enabled then
        local humanoid = char:FindFirstChildOfClass("Humanoid")
        if humanoid then
            humanoid.AutoRotate = false
        end
        if rotationConnection then
            rotationConnection:Disconnect()
            rotationConnection = nil
        end
        startRotation()
    end
end)

tRotation:AddSection({ Name = "旋转功能" })

tRotation:AddToggle({

    Default = false,
    Callback = function(v)
        RotationConfig.Enabled = v

        if hrp then
            hrp.Anchored = true
        end

        if humanoid then
            local animator = humanoid:FindFirstChildOfClass("Animator")
            if animator then
                for _, track in ipairs(animator:GetPlayingAnimationTracks()) do
                    if track.Speed ~= 0 then
                        cachedAnimationSpeeds[track] = track.Speed
                        track:AdjustSpeed(0)
                    end
                end
            end
        end
    end
end)

local CreatorESPObjects = {}
local creatorInGame = false

local function removeCreatorESP(player)
    if player and player.Character then
        local esp = player.Character:FindFirstChild("CreatorESP")
        if esp then esp:Destroy() end
        local highlight = player.Character:FindFirstChild("CreatorHighlight")
        if highlight then highlight:Destroy() end
    end
    if CreatorESPObjects[player] then
        CreatorESPObjects[player] = nil
    end
end

local function createCreatorESP(player)
    if not player or not player.Character or not player.Character:FindFirstChild("HumanoidRootPart") then
        return
    end
    if CreatorESPObjects[player] and CreatorESPObjects[player].ESP.Parent then
        return
    end
    removeCreatorESP(player)

    local esp = Instance.new("BillboardGui")
    esp.Name = "CreatorESP"
    esp.Size = UDim2.new(0, 200, 0, 50)
    esp.StudsOffset = Vector3.new(0, 3, 0)
    esp.AlwaysOnTop = true
    esp.MaxDistance = 1000

    local label = Instance.new("TextLabel")
    label.Name = "CreatorESPText"
    label.Size = UDim2.new(1, 0, 1, 0)
    label.BackgroundTransparency = 1
    label.Text = "BKU创建者"
    label.TextColor3 = Color3.fromRGB(0, 150, 255)
    label.TextScaled = false
    label.TextSize = 14
    label.Font = Enum.Font.GothamBold
    label.Parent = esp

    esp.Parent = player.Character.HumanoidRootPart

    local highlight = Instance.new("Highlight")
    highlight.Name = "CreatorHighlight"
    highlight.FillColor = Color3.fromRGB(0, 150, 255)
    highlight.FillTransparency = 1
    highlight.OutlineColor = Color3.fromRGB(0, 150, 255)
    highlight.OutlineTransparency = 0
    highlight.Parent = player.Character

    CreatorESPObjects[player] = {
        ESP = esp,
        Highlight = highlight,
    }
end

local function findAndMarkCreator()
    local foundCreator = false
    for _, player in pairs(Players:GetPlayers()) do
        if player.Name:lower() == "baloksxin_real" then
            foundCreator = true
            if player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
                if not CreatorESPObjects[player] or not player.Character:FindFirstChild("CreatorESP") then
                    createCreatorESP(player)
                    if not creatorInGame then
                        OrionLib:MakeNotification({ Name = "BKU SCRIPT", Content = "BKU创建者已进入此服务器", Time = 6 })
                        creatorInGame = true
                    end
                end
            end
        else
            if CreatorESPObjects[player] then
                removeCreatorESP(player)
            end
        end
    end
    if not foundCreator and creatorInGame then
        creatorInGame = false
    end
end

Players.PlayerAdded:Connect(function(player)
    if player.Name:lower() == "baloksxin_real" then
        OrionLib:MakeNotification({ Name = "BKU SCRIPT", Content = "BKU创建者已进入此服务器", Time = 6 })
        creatorInGame = true
        player.CharacterAdded:Connect(function()
            task.wait(0.5)
            removeCreatorESP(player)
            createCreatorESP(player)
        end)
        if player.Character then
            task.wait(1)
            createCreatorESP(player)
        end
    end
end)

Players.PlayerRemoving:Connect(function(player)
    if player.Name:lower() == "baloksxin_real" then
        removeCreatorESP(player)
        creatorInGame = false
    end
end)

task.spawn(function()
    task.wait(1)
    findAndMarkCreator()
    while true do
        findAndMarkCreator()
        task.wait(2)
    end
end)
